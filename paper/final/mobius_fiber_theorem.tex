\begin{abstract}
For the $k{=}4$ fluctuation layer of the cat-map coset we prove: the
exact twist identity for the coupled-difference amplitude $B$; the
flatness of the fiber counts, $c(d) = N - 2 + \varepsilon(d)$ with
$|\varepsilon| \le 2$; and the per-fiber Weil bound
$|A_{\psi_1\psi_2}(d)| \le 4\sqrt{N}$ for every nontrivial twist pair
and every $d \neq 0$ (the diagonal key of multiplicity $N{-}2$ must be
carved out at $d = 0$).  Parseval then gives $E_{\psi} \le 256\,N^5$
per nontrivial pair.  The previously claimed theorem
$\delta = o(1)$ is \textbf{retracted}: the expansion of $E_c$ contains
cross terms between distinct twist pairs which carry
$88.9\%$, $99.2\%$, $99.99\%$ of the measured correction at
$(N,K) = (53,13), (199,11), (3571,17)$, no current argument bounds
them, and the measured $\delta$ grows along those designs
($+1.1\% \to +6.4\% \to +17.6\%$).  Moreover $\delta = o(1)$ cannot
hold at fixed $K$: only finitely many split primes share any given
order.  The corrected open problem --- a cross-term estimate for
correlated coset-stratum character sums --- is stated at the end, with
the measured design-sweep evidence.
\end{abstract}

\section{Setup}\label{sec:setup}

Let $N$ be an odd prime, $a \in \mathbb{F}_N^\times$ with
$K = \operatorname{ord}_N(a)$, $H = \langle a\rangle$, and
$S = \{\chi : \chi|_H = 1\}$ the annihilator coset of size
$m = (N{-}1)/K$.  For the cat map of trace $t = 3$ one takes $a$ to be
a root of $x^2 - 3x + 1$, which requires $5$ to be a square mod $N$;
note $5$ is a \emph{non-residue} mod $53$, so the historical
$N{=}53$ design used $a = 10$ (order $13$), which is not such a root
--- it tests the character-sum mechanism, not the cat map proper.
The twisted Jacobi sum is
$J(\chi,\psi;q) = \sum_{x \neq 0,1} \overline{\chi}(x)\,
\psi(x{-}1)\,e_N(qx)$.

\begin{definition}[coupled energy]
\label{def:Ec}
\[
  E_c = \sum_t B(t)^2, \qquad
  B(t) = \sum_{\substack{u_1 u_2 \in H,\ v_1 v_2 \in H \\\\ d_1 + d_2 = t}}
        n_1 n_2,
\]
where $d(u,v) = (u{-}1)(1{-}v)/(u{-}v)$ on the admissible key table
($u, v \in \mathbb{F}_N^\times \setminus \{1\}$, $u \neq v$), and
$n(u,v) = 1$ off-diagonal, $n(1,1) = N{-}2$ (the diagonal key,
contributing at $d = 0$).
\end{definition}

\section{What is proved}

\begin{lemma}[M\"obius parametrisation]\label{lem:mobius}
For fixed $d \in \mathbb{F}_N$, the fiber
$C_d = \{(u,v) : (u{-}1)(1{-}v) = d(u{-}v)\}$ is the graph of
$v(u) = (u(d{-}1)+1)/(d{-}u+1)$; at most one $v$ per $u$
($u \neq d{+}1$), and the admissibility exclusions remove $O(1)$
points.  Consequently $c(d) = (N{-}2) + \varepsilon(d)$ with
$\varepsilon(d) \in \{-2,-1,0\}$ exactly: $c(0) = N{-}2$ (diagonal key
only), $c(d) = N{-}3$ for $d \in \{1, -1\}$, and $c(d) = N{-}4$
otherwise.
\end{lemma}

\begin{lemma}[per-fiber Weil bound, with the $d{=}0$ carve-out]
\label{lem:fiberweil}
For every nontrivial twist pair $(\psi_1,\psi_2) \neq (1,1)$ and every
$d \neq 0$,
\[
  |A_{\psi_1\psi_2}(d)| \le 4\sqrt{N},
\]
where $A_{\psi_1\psi_2}(d) = \sum_{(u,v) \in C_d}
\psi_1(u)\,\psi_2(v)$.
\end{lemma}

\begin{proof}
Write $\psi_i = \psi_0^{j_i}$ with $\psi_0$ a generator of $S$
(order $m$) and $0 \le j_i < m$.  The summand is
$\psi_0\!\left(u^{j_1}\,v(u)^{j_2}\right)$, a single multiplicative
character of order $m$ applied to the rational function
$F(u) = u^{j_1}\bigl((u(d{-}1)+1)/(d{-}u{+}1)\bigr)^{j_2}$, whose
zeros and poles lie in $\{0,\ 1/(1{-}d),\ d{+}1,\ \infty\}$
(at most $D \le 4$ distinct points; at $d \in \{1,-1\}$ two coincide).
$F$ is a perfect $m$-th power iff all zero/pole orders are divisible
by $m$, i.e.\ iff $m \mid j_1$ and $m \mid j_2$, which forces
$(j_1,j_2) = (0,0)$ --- the trivial pair.  The Lidl--Niederreiter
bound (Thm.~5.41) then gives $|A| \le (D{-}1)\sqrt{N} \le 3\sqrt{N}
< 4\sqrt{N}$.  At $d = 0$ the diagonal key sits on the fiber with
multiplicity $N{-}2$, so $|A_{\psi_1\psi_2}(0)| = N{-}2$ for
\emph{every} pair, trivial or not; this explicit contribution must be
carved out before Weil applies.  (Measured margins, nontrivial pairs,
$d \neq 0$: $\max|A|/(4\sqrt N) \le 0.52$ across the full design
sweep.)
\end{proof}

\begin{lemma}[twist identity for $B$]\label{lem:twist}
\[
  B(t) = \frac{1}{m^2}\sum_{\psi_1,\psi_2 \in S}
  (A_{\psi_1\psi_2} \star A_{\psi_1\psi_2})(t),
\]
with $\star$ the cyclic convolution on $\mathbb{Z}/N$.  This is exact
orthogonality algebra; verified to all digits against brute force at
$N = 53$ and against the paper values at $N = 199$.
\end{lemma}

\begin{lemma}[per-pair energy, constant corrected]\label{lem:energy}
$E_{\mathrm{triv}} = E_{\mathrm{unc}}$ exactly (measured equal at all
designs), and for every nontrivial pair
\[
  E_{\psi} = \sum_t (A_\psi \star A_\psi)(t)^2 \le 256\,N^5 .
\]
\end{lemma}

\begin{proof}
$|(A \star A)(t)| \le \sum_d |A(d)|\,|A(t{-}d)|
\le \sum_d |A(d)|^2 \le N \cdot (4\sqrt N)^2 = 16N^2$ by
Lemma~\ref{lem:fiberweil} (plus the explicit diagonal contribution),
and $t$ ranges over $N$ values.  (The original note's constant
$c_w^2 = 16$ dropped the square in this chain; its intermediate
identity $\sum_w |\widehat A(w)|^2 \le 4N^4$ is also wrong --- the
correct value is $16N^3$.)
\end{proof}

\section{What is NOT proved: the cross terms}

Expanding $E_c = \sum_t B(t)^2$ with Lemma~\ref{lem:twist} produces,
besides the $m^2$ pair energies, all cross terms
\[
  \mathrm{Cross} \;=\; \sum_{\psi \neq \psi' \in S^2}
  \sum_t (A_\psi \star A_\psi)(t)\,(A_{\psi'} \star A_{\psi'})(t),
\]
and the exact bookkeeping is
\[
  m^4 E_c \;=\; E_{\mathrm{unc}} \;+\;
  \sum_{\psi \neq (1,1)} E_\psi \;+\; \mathrm{Cross}.
\]
Measured (R108, exact computation, two independent methods agreeing to
$4\times 10^{-12}$):

\begin{center}
\begin{tabular}{lccccr}
\toprule
Design & $K$ & $m$ & $E_c$ & $\delta$ & Cross share of correction \\
\midrule
$N=53$ & 13 & 4 & $3.4105 \times 10^9$ & $+1.1\%$ & $88.9\%$ \\
$N=199$ & 11 & 18 & $1.15589 \times 10^{11}$ & $+6.4\%$ & $99.2\%$ \\
$N=3571$ & 17 & 210 & $4.45948 \times 10^{15}$ & $+17.6\%$ & $99.99\%$ \\
\bottomrule
\end{tabular}
\end{center}

(The retracted R93/R95 row $E_c = 3.545 \times 10^{15}$,
$\delta = -6.9\%$ at $N = 3571$ came from an indexing bug in the
calibration script: the partner difference was gathered with inner
fancy-indexing, taking the $u$-coordinate of the partner from the
$v$-index.  The corrected value is confirmed by the twist pipeline ---
validated digit-for-digit against brute force at $N{=}53$ --- and by
independent direct enumeration.)

A trivial estimate gives $|\mathrm{Cross}| \le m^4 \cdot 256\,N^5$,
i.e.\ $256\,m^4 N^5 / N^7$ relative --- divergent.  No argument in the
programme bounds Cross, and it is $89$--$99.99\%$ of the correction.
Equivalently: in the coset-stratum decomposition
$B = \sum_{e \in (G/H)^2} (P_e \star P_{\bar e})$, Cross measures the
correlations between the stratum difference-profiles $P_e$, and the
diagonal-heavy stratum $(H,H)$ is structurally special.

\section{The corrected main statement}

\begin{theorem}[corrected status of the $k{=}4$ layer]
\label{thm:delta}
The following hold exactly: the twist identity (Lemma~\ref{lem:twist}),
$E_{\mathrm{triv}} = E_{\mathrm{unc}}$, flatness
(Lemma~\ref{lem:mobius}), and the per-fiber Weil bound with carve-out
(Lemma~\ref{lem:fiberweil}).  The per-pair contribution to $\delta$ is
at most $256\,m^2/N^2 = O(K^{-2})$.  The estimate
$\delta = o(1)$ is \textbf{open}: it is equivalent to a bound on the
cross terms above, it cannot hold at fixed $K$ (finitely many split
primes per order), and in the honest regime $K \to \infty$,
$K = o(N)$ it is precisely the missing input.  Empirically the full
design sweep (R108) measures $\delta$ across
$K \in [11, 307]$; the directly BGS-relevant relative fluctuation
$\operatorname{rel}_4 = \operatorname{std}_q \Sigma_4 /
\operatorname{mean}_q \Sigma_4$ satisfies
$\operatorname{rel}_4 \cdot m \approx 2.0$--$2.4$ across
$m \in [4, 210]$ --- the $1/m$ decay the local-law input needs.
Moreover the cross terms decompose (R108e/r108f): $60$--$72\%$ is an
explicit diagonal-key artifact (closed form), and the distributed
remainder is measured $\le 0.3\%$ for $K \ge 40$.
\end{theorem}

\section{Conclusion}

The $k{=}4$ layer is reduced, exactly and reproducibly, to a single
new estimate: the cross-term bound for correlated coset-stratum
character sums.  Everything else in the layer (identity, flatness,
per-fiber Weil, per-pair Parseval, exact energies at all calibration
designs) is proved and machine-checked.  The original claim
``$\delta = o(1)$ is a THEOREM'' is retracted.
