\title{Exact Character Reductions and an Elliptic-Family Route\\to Fourth-Moment Concentration for Arithmetic Quantum Cat Maps}
\author{Maksim Churilov}
\date{Research draft --- 3 September 2026}

\begin{document}
\maketitle

\begin{abstract}
We isolate an exact finite-field object controlling a character-averaged fourth moment arising in arithmetic quantum cat maps. The construction replaces an earlier invalid M\"obius-fiber assembly by a weighted pair-key distribution $B(t)$. We prove exact formulas for the pair parametrization, the total fourth-moment constraint count, the character-orthogonality collapse, and a Fourier variance duality. The mixed term is transformed by a Cayley map into an additive-energy problem; for fixed odd prime subgroup order $K$, outside an explicitly computable finite set of reduction primes, its energy is the forced central-symmetry baseline. The remaining off/off term is organized by a non-isotrivial elliptic family. We state precisely the algebraic and sheaf-theoretic obligations still required to promote the resulting $K=o(N^{1/4})$ fourth-moment estimate to a theorem. Finite GOE-like spectral data are numerical evidence only. No local BGS eigenphase law is claimed.
\end{abstract}

\section{Scope}
The Bohigas--Giannoni--Schmit principle predicts random-matrix local spectral statistics for quantizations of classically chaotic systems after exact symmetries are removed. Arithmetic cat maps are delicate because Hecke symmetries and finite-field algebra strongly influence both eigenfunctions and spectra. Quantum ergodicity or a matrix-element moment theorem does not by itself imply a microscopic GOE/GUE eigenphase law. The present paper concerns an exact finite-field reduction and a route to one fourth-moment concentration estimate; the local spectral-law step remains open.

\section{Pair-key geometry}
Let $N$ be an odd prime and
\[
H\leq\F_N^\times,\qquad |H|=K,\qquad m=\frac{N-1}{K}.
\]
Let $S=H^\perp$ be the annihilator subgroup of multiplicative characters. For $x,y\in\F_N\setminus\{0,1\}$ put
\[
u=\frac{x}{y},\qquad v=\frac{x-1}{y-1},\qquad d=x-y.
\]

\begin{proposition}[Pair-key parametrization]
If $x\ne y$, the map $(x,y)\mapsto(u,v)$ is a bijection onto
\[
D_{\rm off}=\{(u,v)\in(\F_N^\times\setminus\{1\})^2:u\ne v\},
\]
with inverse
\[
y=\frac{1-v}{u-v},\qquad x=\frac{u(1-v)}{u-v},
\]
and
\[
d(u,v)=\frac{(u-1)(v-1)}{v-u}.
\]
On the diagonal, all $N-2$ admissible pairs have key $(1,1)$ and $d=0$.
\end{proposition}

The diagonal multiplicity is load-bearing. Replacing this weighted table by the historical M\"obius graph changes the fourth-moment object.

\section{Exact fourth-moment reduction}
A collapsed key record is $r=(u_r,v_r,d_r,n_r)$, where $n_r$ is its multiplicity. Define
\[
B(t)=\sum_{r,s}n_rn_s\,\1_H(u_ru_s)\,\1_H(v_rv_s)\,\1_{d_r+d_s=t}.
\]
The $k=2$ constrained off-diagonal pair count is exactly $(K-1)(K-2)$.

\begin{theorem}[Exact $k=4$ total]
\[
\sum_tB(t)=(K^2+1)N^2-(6K^2+3K)N+13K^2-3K.
\]
\end{theorem}

\begin{proof}
The diagonal/diagonal class contributes $(N-2)^2$, and each ordered diagonal/off-diagonal class contributes $(N-2)(K-1)(K-2)$. In the off/off class a partner of $(u,v)$ is $(h_1/u,h_2/v)$ with $(h_1,h_2)\in H^2$. Failure to remain in $D_{\rm off}$ occurs only through $h_1=u$, $h_2=v$, or $h_1v=h_2u$. Inclusion--exclusion gives
\[
V_{\rm off}=K^2(N-2)(N-3)-3K(K-1)(N-3)+2(K-1)(K-2),
\]
and summing the four ordered classes yields the formula.
\end{proof}

For $\chi,\psi\in S$, define
\[
A_{\chi,\psi}(d)=\sum_{r:d_r=d}n_r\chi(u_r)\psi(v_r).
\]
Character orthogonality gives the exact identity
\[
\sum_{\chi,\psi\in S}(A_{\chi,\psi}*A_{\chi,\psi})(t)=m^2B(t).
\]
For the twisted Jacobi sums
\[
J(\chi,\psi;q)=\sum_{x\in\F_N\setminus\{0,1\}}\overline{\chi(x)}\,\psi(x-1)\eN{qx},
\]
this becomes
\[
\sum_{\chi,\psi\in S}|J(\chi,\psi;q)|^4=m^2\Sigma_4(q),\qquad
\Sigma_4(q)=\sum_tB(t)\eN{qt}.
\]

Writing $V=\sum_tB(t)$ and $Z=B(0)$, finite Fourier orthogonality gives
\[
\E_{q\ne0}\Sigma_4(q)=\frac{NZ-V}{N-1},
\qquad
\Var_{q\ne0}\Sigma_4(q)=N\Var_{t\ne0}B(t).
\]
Thus $\operatorname{std}_{q\ne0}\Sigma_4(q)=O(NK)$ is exactly equivalent to $\operatorname{std}_{t\ne0}B(t)=O(K\sqrt N)$.

\section{The mixed Cayley term}
There is an exact decomposition
\[
B(t)=O(t)+2(N-2)C_H(t)+(N-2)^2\1_{t=0}.
\]
For $h\ne1$ define $c(h)=(h+1)/(h-1)$. Then
\[
d(u,v)=\frac{2}{c(u)-c(v)}.
\]
Hence $C_H$ is a relabelled nonzero difference-representation function of $A=c(H\setminus\{1\})$. Since $c(h^{-1})=-c(h)$, for odd $K$ the forced collision energy is
\[
(K-1)(2K-5).
\]

For odd prime $K$, clearing denominators in a Cayley collision produces a sparse cyclotomic relation. An exact finite classification of the $6!$ exponent matchings leaves only identity and inversion-swap families in characteristic zero. Reduction modulo $N$ can introduce additional collisions. For fixed $K$, define $E_K$ from the prime divisors, congruent to $1\pmod K$, of the corresponding nonzero cyclotomic resultants.

\begin{proposition}[Fixed-$K$ mixed energy]
If $K$ is an odd prime and $N\equiv1\pmod K$ is prime with $N\notin E_K$, then
\[
\sum_tC_H(t)^2=(K-1)(2K-5).
\]
\end{proposition}

The active exact gate fully enumerates
\[
E_{11}=\{23,67,89\},\qquad E_{13}=\{53,79,131,157,313\}.
\]
This is not a useful uniform exclusion when $K$ grows. A negative control is $N=461,K=23$, where direct exact computation gives mixed energy $1190$ rather than the forced baseline $902$.

Outside $E_K$ the mixed term $2(N-2)C_H(t)$ has standard deviation $O(K\sqrt N)$.

\section{Off/off elliptic family}
After Cayley coordinates, the pure off/off term can be organized by a bounded-degree family of plane cubics. Current symbolic computations support a nonconstant generic $j$-invariant, bounded bad-fiber complexity, an involutive same-$j$ symmetry, and reducible fibers at Cayley-difference parameters already detected by $C_H$.

This structure suggests a trace-sheaf correlation argument, but two load-bearing obligations remain before an unconditional theorem can be stated:

\begin{enumerate}
\item a publication-grade projective/Weierstrass derivation of the family, its discriminant and $j$ map, complete exceptional set, and the required no-isogeny/non-isomorphism statement after reduction, including twists and inseparable exceptional cases;
\item an exact uniform application of a trace-sheaf quasi-orthogonality theorem with purity, geometric irreducibility, monodromy, conductor, ramification, dual/twist and deleted-point hypotheses checked for the actual family.
\end{enumerate}

\begin{proposition}[Conditional restricted M1 bound]
Assume these two obligations with constants uniform in $N$ and in the allowed family parameters. Then
\[
\Var_{t\ne0}O(t)\ll K^2N+K^4N^{1/2}+K^4.
\]
Consequently, outside the relevant fixed-$K$ exceptional reductions and in the preliminary regime $K=o(N^{1/4})$,
\[
\Var_{t\ne0}B(t)=O(K^2N),\qquad
\operatorname{std}_{q\ne0}\Sigma_4(q)=O(NK).
\]
\end{proposition}

\begin{remark}
The proposition is conditional. It is not promoted to a theorem in the current repository.
\end{remark}

\section{Finite verification and spectral evidence}
Independent finite implementations reproduce the anchors
\[
\begin{array}{c|c|r|r|r}
N&K&\sum_tB(t)&B(0)&\sum_tB(t)^2\\\hline
23&11&48621&2609&103156981\\
53&13&423879&12063&3410502105\\
199&11&4681821&96641&115589503389\\
461&23&111147915&641799&26960554630421
\end{array}
\]
Finite parity-reduced spectra at $N=3571,K=17$ and $N=19489,K=29$ are compatible with GOE-like spacing on the recorded samples, and number-variance calculations show comparable rigidity. These observations motivate the target but are not proof of a limiting local law.

\section{Open local-statistics step}
Even an unconditional restricted M1 theorem would not finish BGS. The next problem is to identify a valid microscopic comparison mechanism for the structured arithmetic matrix family: its dependence structure, symmetry class, resolvent object and error scale must be explicit before importing Wigner-matrix local-law technology.
