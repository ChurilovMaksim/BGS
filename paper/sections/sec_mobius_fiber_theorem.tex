\section{The M\"obius--fiber theorem: corrected status of $\delta$}
\label{sec:r101}

\subsection{The fiber sums are M\"obius-twisted character sums}

The difference fiber over $d$ is the conic
$(u{-}1)(1{-}v) = d(u{-}v)$, which solves for $v$ as the M\"obius
function of $u$,
\[
  v(u) = \frac{u(d-1)+1}{\,d-u+1\,}.
\]
The twisted fiber sum of Theorem~\ref{thm:twist} is therefore a sum of
a multiplicative character evaluated along a rational parametrization:
\[
  A_{\psi_1\psi_2}(d) = \sum_{u \ \mathrm{adm.}} \psi_1(u)\,
  \psi_2\!\left(\frac{u(d-1)+1}{d-u+1}\right).
\]

\subsection{The per-fiber Weil bound, with the $d=0$ carve-out}

\begin{theorem}[per-fiber bound; $d=0$ carve-out added in R108]
\label{thm:fiberweil}
For a nontrivial twist pair $(\psi_1,\psi_2) \neq (1,1)$ and every
$d \neq 0$,
\[
  |A_{\psi_1\psi_2}(d)| \le c_w \sqrt{N}, \qquad c_w = 4,
\]
by the Weil bound for multiplicative character sums with rational
functions (Lidl--Niederreiter, \emph{Finite Fields}, Thm.~5.41: the
integrand $\psi_1(u)\,\psi_2\!\bigl((u(d{-}1)+1)/(d{-}u{+}1)\bigr)$ is
a single character of order $m$ applied to
$F(u) = u^{j_1}\bigl((u(d{-}1)+1)/(d{-}u{+}1)\bigr)^{j_2}$, whose zeros
and poles lie in the at-most-4-point set
$\{0,\,1/(1{-}d),\,d{+}1,\,\infty\}$, and the sum is bounded by
$(D-1)\sqrt{N} \le 3\sqrt{N}$ unless $F$ is a perfect $m$-th power,
which happens only for the trivial twist).  At $d = 0$ the diagonal key
$(1,1)$ of multiplicity $N{-}2$ sits on the fiber, so
$|A_{\psi_1\psi_2}(0)| = N-2$ for \emph{every} pair; this contribution
is explicit and must be carved out before applying Weil.  Measured
(nontrivial pairs, $d \neq 0$): $\max |A|/(4\sqrt N) \le 0.52$ across
the whole R108 design sweep.
\end{theorem}

\begin{theorem}[the $\delta$ bound; OPEN after R108]
\label{thm:deltabound}
The per-fiber bound gives, by Parseval, $E_{\psi} \le 256\,N^5$ per
nontrivial pair (constant corrected from 16 in R108: the original
chain $|(A \star A)(t)|$-estimate yields $16N^2$ per $t$, hence
$256\,N^5$ per pair), contributing at most $256\,m^2/N^2 = O(K^{-2})$
to $\delta$.  However, the expansion of $E_c$ also contains cross
terms between distinct twist pairs, which carry $88.9\%$--$99.99\%$ of
the measured correction $m^4 E_c - E_{\mathrm{unc}}$ and are bounded by
no current argument.  Moreover $\delta = o(1)$ cannot hold at fixed
$K$ (only finitely many split primes share a given order).
Conjecture~\ref{conj:Ec} at $k{=}4$ is therefore \textbf{OPEN}.
\end{theorem}

\subsection{The exact pair-type decomposition (R108g, new)}

\begin{theorem}[pair-type decomposition; exact]
\label{thm:pairtype}
Splitting the coupled key pairs by type,
\[
  B(t) = (N{-}2)^2\,\delta_{t=0} \;+\; 2(N{-}2)\,p_H(t) \;+\;
  B_{\mathrm{off}}(t),
\]
with $p_H(t) = \#\{\text{off-keys at difference } t \text{ with }
u, v \in H\}$ (a $K^2$-sized explicit table) and $B_{\mathrm{off}}$
the coupled amplitude of the diagonal-free system.  Hence
\[
  E_c = (N{-}2)^4 + 2(N{-}2)^2 B_{\mathrm{off}}(0)
  + 4(N{-}2)^2 \textstyle\sum p_H^2
  + 4(N{-}2)\!\!\sum_t p_H B_{\mathrm{off}} + E_c^{\mathrm{off}}.
\]
Machine-verified to $0.00$ relative error at every design.  The
mechanism: the diagonal spike is twist-\emph{invariant} (its fiber
contribution is $(N{-}2)\,\psi(1,1) = N{-}2$ for every twist), so it
survives the annihilator averaging at full weight while the flat
baseline $E_{\mathrm{unc}}/m^4$ suppresses the background by $m^4$;
the measured $\delta$ is this mismatch.  The clean object is
$\delta_{\mathrm{off}} := m^4 E_c^{\mathrm{off}}/E_{\mathrm{unc}}^{\mathrm{off}} - 1
\approx \delta/4$--$5$, measured $\le 0.2\%$ for $K \ge 40$.
\end{theorem}
