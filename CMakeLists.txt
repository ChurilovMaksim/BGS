cmake_minimum_required(VERSION 3.20)
project(wow_catmap_verifiers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 NEW)
endif()
find_package(Boost REQUIRED)
add_executable(fourth_moment_pair_key_verifier computations/cpp/fourth_moment_pair_key_verifier.cpp)
add_executable(cat_map_order computations/cpp/cat_map_order.cpp)
add_executable(cayley_energy_verifier computations/cpp/cayley_energy_verifier.cpp)
add_executable(cayley_fibonacci_verifier computations/cpp/cayley_fibonacci_verifier.cpp)
add_executable(cayley_direct_energy_verifier computations/cpp/cayley_direct_energy_verifier.cpp)
add_executable(cayley_trace_intercept_rigidity computations/cpp/cayley_trace_intercept_rigidity.cpp)
add_executable(fixed_strength_locality_exact computations/cpp/fixed_strength_locality_exact.cpp)
add_executable(magnetic_word_verifier computations/cpp/magnetic_word_verifier.cpp)
add_executable(pell_coboundary_pairs computations/cpp/pell_coboundary_pairs.cpp)
add_executable(pell_pair_orthogonality computations/cpp/pell_pair_orthogonality.cpp)
add_executable(pell_floquet_pair_verifier computations/cpp/pell_floquet_pair_verifier.cpp)
add_executable(spectral_lattice_clustering computations/cpp/spectral_lattice_clustering.cpp)
add_executable(pell_minimal_relation_verifier computations/cpp/pell_minimal_relation_verifier.cpp)
add_executable(general_hyperbolic_periods computations/cpp/general_hyperbolic_periods.cpp)
add_executable(pell_sharp_orbit_verifier computations/cpp/pell_sharp_orbit_verifier.cpp)
target_include_directories(pell_coboundary_pairs PRIVATE ${Boost_INCLUDE_DIRS})
target_include_directories(pell_pair_orthogonality PRIVATE ${Boost_INCLUDE_DIRS})
target_include_directories(pell_floquet_pair_verifier PRIVATE ${Boost_INCLUDE_DIRS})
target_include_directories(spectral_lattice_clustering PRIVATE ${Boost_INCLUDE_DIRS})
target_include_directories(general_hyperbolic_periods PRIVATE ${Boost_INCLUDE_DIRS})
target_include_directories(pell_sharp_orbit_verifier PRIVATE ${Boost_INCLUDE_DIRS})
target_include_directories(cayley_fibonacci_verifier PRIVATE ${Boost_INCLUDE_DIRS})
foreach(target fourth_moment_pair_key_verifier cat_map_order cayley_energy_verifier cayley_fibonacci_verifier cayley_direct_energy_verifier cayley_trace_intercept_rigidity fixed_strength_locality_exact magnetic_word_verifier pell_coboundary_pairs pell_pair_orthogonality pell_floquet_pair_verifier spectral_lattice_clustering pell_minimal_relation_verifier general_hyperbolic_periods pell_sharp_orbit_verifier)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
