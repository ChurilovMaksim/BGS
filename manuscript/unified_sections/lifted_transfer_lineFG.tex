\begin{theorem}[Lifted trace identity, BGS-TRACE-197, proved]
Let \(\rho_V>\lambda/(2(\lambda-1))\approx0.6036\), \(\rho_E>1/(2(\lambda-1))\approx0.1036\), \(\beta>0\), and \(\sum_n\lvert w(n)\rvert e^{\alpha\lvert n\rvert}<\infty\) with \(\alpha>2\pi c\max(\rho_V,\rho_E)+\beta/\lambda\). The vertex and edge operators \(L_V,L_E\) on weighted \(\ell^2\)-sums of Hardy spaces over the disc lattices are trace class, and for every \(m\ge1\), \(\sum_{n\in\mathbb Z^m}\prod_jw(n_j)e(cn_js_j)=(1-\lambda^{-m})(\operatorname{Tr}L_V^m-\operatorname{Tr}L_E^m)\), where \(s_j\) is the periodic orbit of the word \(n\). Hence \(Z_m(\nu^*)=(1-\lambda^{-m})\sum_{a\in\{0,2,4,6\}}c_a(\operatorname{Tr}L_{V,a}^m-\operatorname{Tr}L_{E,a}^m)\) for the full alphabet. The counting step (number of closed vertex chains minus closed edge chains equals one) is formalised in Lean.
\end{theorem}

\begin{theorem}[Certified spectrum at eta two, BGS-TRACE-198, proved (computer-assisted)]
For the full-alphabet \(k1\) operators at \(\eta=2\), \(L_{V,4}\) has exactly one eigenvalue \(\mu^*\) (simple) with \(\lvert\mu^*\rvert\ge1\), and \(\lvert\mu^*\rvert\in[1.9154712,1.9154753]\); all other spectrum of all sectors of both lattices lies in \(\lvert z\rvert<1\); \(\lVert L_V\rVert_1\le730.15\), \(\lVert L_E\rVert_1\le22.94\). The reflection \((U\Phi)_k(s)=\Phi_{-k}(-s)\) is unitary and conjugates sector 2 to sector 6.
\end{theorem}

\begin{theorem}[All-m failure of the k1 analogue of Conjecture R, BGS-TRACE-199, proved (computer-assisted)]

\end{theorem}

\begin{theorem}[General partial-trace law, BGS-TRACE-203, proved (finite alphabet)]
For every unit \(\nu\) and every \((a,b)\) with \(b\nu\equiv a\pmod{2h}\), and every finitely supported kick, \(Z_m(\nu)=(1-\lambda^{-m})\operatorname{Tr}_H[(I\otimes\langle\mathbb 1\rvert)\mathcal L^m(I\otimes\lvert\delta_0\rangle)]\) with \(\mathcal L=\sum_n\hat g(n)R_n\otimes A_n\) on \(H^2(D_\rho)\otimes\mathbb C^{b^2}\), \(R_n\) a weighted composition operator with coupling \(-4ac/b\) and \(A_n\) the automaton step for \(P\bmod b\). On each residue class of \(m\) the sequence is an exponential sum over the eigenvalues of a fixed trace-class operator.
\end{theorem}
