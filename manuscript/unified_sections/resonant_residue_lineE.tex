Notation is that of Section~\ref{sec:derandomized-fixed-kick}: \(Z_m(\nu)=\sum_{n\in\mathbb Z^m}\hat\Phi(n)e(-\nu y'_m\operatorname{Nm}(\alpha_n)/g'_m)\),

\paragraph{Scope and non-claims.}
\begin{enumerate}[label=(\arabic*)]
\item Three kicks occur: \(k1\), \(f(q)=0.9\cos2\pi q\); \(f_7\), \(f(q)=0.9\cos2\pi\cdot7q\); and the Line B diagonal kick \(f_B\) (14 modes, \(c_{\pm7}=9/20\)), which is the kick of Conjecture R (BGS-TRACE-179). Every rigorous growth statement for all large \(m\) concerns \(k1\) (finite range, or conditional).
\item Within this section Conjecture R is not disproved for any kick: finitely many \(m\) cannot contradict an \(o(m)\) statement, and the asymptotic statements are conditional. BGS-TRACE-180 and 182 stay open. (Later: BGS-TRACE-179 is refuted as stated at \(\eta=2\) by BGS-TRACE-209, Section~\ref{sec:lineH-49state}.)
\item Nothing here is a counterexample to BGS, nothing concerns eigenvalue statistics beyond the second moment of \(\operatorname{Tr}U^m\), and there is no progress on BGS-TRACE-148.
\item The starting formula is BGS-TRACE-163, Theorem 1 (Boasman--Keating). The resonance is analogous to, but not shown to be an instance of, a Keating--Mezzadri pseudo-symmetry.
\end{enumerate}

\begin{theorem}[Resonant residue, BGS-TRACE-183, proved]
Let \(m\ge5\) be odd and \(\nu^*\) the odd lift of \(-\bar{4}\bmod h_m\) (\(N\equiv-4\bmod h_m\)). For every kick with \(\sum n^2\lvert\hat g(n)\rvert<\infty\),
\(Z_m(\nu^*)=\sum_n\hat\Phi(n)\,e\bigl(\tfrac12q(n)+\kappa_m(\textstyle\sum_jn_j)^2/8\bigr)\) with \(\kappa_m\) odd; at finite odd \(N\equiv-4\bmod h_m\), \(c\mapsto c(1+4/N)\) and \(\lvert\operatorname{Tr}M_N^m\rvert=1\).
\end{theorem}

\begin{theorem}[IFS law, BGS-TRACE-184, proved]
If \(\hat g\) is supported in \(\lvert n\rvert\le L\) and \(\rho>L/(\lambda-1)\), then \(Z_m(\nu^*)=(1-\lambda^{-m})\sum_{a\in\{0,2,4,6\}}c_a\operatorname{Tr}\mathcal L_a^m\),
\(\lvert c_a\rvert=\tfrac12\), with \((\mathcal L_a\varphi)(s)=\sum_{\lvert n\rvert\le L}\hat g(n)e(cn^2/2+an/8)e(cns)\varphi((s+n)/\lambda)\) on \(H^2(D_\rho)\), independent of \(m\).
\end{theorem}

\begin{theorem}[Certified spectrum and growth for the \(k1\) kick, BGS-TRACE-185 to 187, proved (computer-assisted)]
For the \(L\)-truncated \(k1\) operators (\(L=6,8\)) at \(\eta\in\{1.25,\dots,3\}\), \(\mathcal L_4\) has a simple eigenvalue \(\mu^*\) (e.g.\ \(\lvert\mu^*\rvert\in[1.9154750,1.9154771]\) at \(\eta=2\), \(L=6\)) and all other spectrum lies in \(\lvert z\rvert\le r_2\) (185); hence \(\lvert Z_L(m)\rvert>0\) grows exponentially for all odd \(m\ge m_0\) (186).
For the physical \(k1\) kick, \(\lvert\tilde W^{\mathrm{rest}}_m\rvert>m\) for odd \(m\in[17,39]\) at \(\eta=2\), \(m\in[19,27]\) at \(\eta=1.75\), \(m\in[23,33]\) at \(\eta=1.5\) (187). This concerns the \(k1\)-kick analogue of Conjecture R only.
\end{theorem}

\begin{theorem}[Conditional asymptotic for \(k1\), BGS-TRACE-188, conditional]
If the full-alphabet \(k1\) operator has an isolated simple eigenvalue \(\lvert\mu_\infty\rvert>\lambda^{1/4}\) dominating the rest of the spectrum (hypothesis \(H_\infty\)), then \(F_m(2)\ge C(\lvert\mu_\infty\rvert^2\lambda^{-1/2})^m\) and the \(k1\)-kick analogue of Conjecture R fails at \(\eta=2\).
\end{theorem}

\begin{theorem}[\(a^2\)-resonance law, BGS-TRACE-194, proved]
Let \(a\) be odd, coprime to \(2h_m\), with \(2\) a square and \(\lambda^m\not\equiv1\) modulo every prime \(p\mid a\); let \(N\equiv-s\,4a^2\bmod h_m\), \(s=\pm1\).
Then the phase of \(Z_m(\nu)\) splits exactly into a real IFS part \((sc/a^2)[\sum n_js_j+\frac12\sum n_j^2]\), a sector term \(\kappa(\sum n)^2/8\) and a part \((\gamma/a^2)[\sum n_j\sigma_j+\bar{2}\sum n_j^2]\) driven by the periodic orbit of \(\sigma\mapsto\ell^{-1}(\sigma+n)\) in \(\mathbb Z/a^2\).
For finite alphabets, \(Z_m(\nu)=(1-\lambda^{-m})\sum_bc_b\operatorname{Tr}\mathcal L_b^m\) with operators on \(H^2(D_\rho)\otimes\ell^2(\mathbb Z/a^2)\) that do not depend on \(m\), because \(\gamma\equiv-s\,\iota(c)=-sr/8\bmod a^2\) for every \(m\) (\(r^2\equiv2\)); only the coefficients \(c_b\) depend on \(m\). For \(a=7\) the hypotheses hold iff \(3\nmid m\).
\end{theorem}

\begin{theorem}[Exact orbit formula and certified bounds for the Line B kick, BGS-TRACE-195, proved (computer-assisted; \(m=11,13\) only)]
For every kick with \(\sum_n\lvert\hat g(n)\rvert<\infty\), odd \(m\) and every unit \(\nu\),
\[\lvert Z_m(\nu)\rvert=h^{-1}\Bigl\lvert\sum_{(z_0,z_1)\in(\mathbb Z/h)^2}e(tR(z)/h)\prod_jg(\tfrac12+z_j/h)\Bigr\rvert,\]
with \(t=(16\nu)^{-1}\bmod h\), \(R(z)=2z^{\mathsf T}Hz/h\bmod h\).
For the Line B kick \(f_B\) at \(\eta=2\) (IEEE kernel with an a-priori rounding bound; reproduced by an independent directed-rounding interval kernel), from the residues \(N\equiv\pm196\) alone: \(F_m/2m\ge3.5221\) and \(\tilde W^{\mathrm{rest}}_m/m\ge3.0442\) at \(m=11\); \(F_m/2m\ge6.4576\) and \(\tilde W^{\mathrm{rest}}_m/m\ge8.9152\) at \(m=13\) (with the residues \(\pm4\) added: \(3.5227\), \(3.0454\), \(6.4579\), \(8.9158\)); at \(m=17\) no bound is claimed (certificate pending; pre-registered predictions \(\lvert Z\rvert^2\approx5.66587\cdot10^8\) and \(7.04767\cdot10^8\) at \(N\equiv\mp196\)). Nothing is claimed as \(m\to\infty\).
\end{theorem}

\begin{theorem}[Conditional failure of Conjecture R for the Line B kick, BGS-TRACE-196, conditional]
If, for some sign \(s\), the full-alphabet 49-state operators of BGS-TRACE-194 for \(f_B\) at \(\eta=2\) and that \(s\) have an isolated simple eigenvalue \(\mu_s\) with \(\lvert\mu_s\rvert>\lambda^{1/4}\) dominating the rest of the spectrum of the \(s\)-family (hypothesis \(H_{\infty,7}(s)\); float evidence \(\lvert\mu_{+1}\rvert\approx1.884\), \(\lvert\mu_{-1}\rvert\approx1.897\), next \(\le1.30\)), then \(\lvert Z_m(\nu_{7,s})\rvert\asymp\lvert\mu_s\rvert^m\), \(F_m(2)\ge C\max_s(\lvert\mu_s\rvert^2\lambda^{-1/2})^m\) over such \(s\), and \(\tilde W^{\mathrm{rest}}_m(2)/m\to\infty\) along odd \(m\) with \(3\nmid m\), so Conjecture R would fail at \(\eta=2\). Nothing is claimed along \(3\mid m\).
\end{theorem}
