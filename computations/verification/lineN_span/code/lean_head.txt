/-- ε^k = (p, q) meaning p + q√2. -/
def epsPowNat : Nat → Int × Int
  | 0 => (1, 0)
  | n + 1 => let (p, q) := epsPowNat n; (p + 2 * q, p + q)
def epsInvPowNat : Nat → Int × Int
  | 0 => (1, 0)
  | n + 1 => let (p, q) := epsInvPowNat n; (-p + 2 * q, p - q)
def epsPow (k : Int) : Int × Int :=
  if 0 ≤ k then epsPowNat k.toNat else epsInvPowNat (-k).toNat

/-- exact sign of u + v√2 (by squares). -/
def sgn2 (u v : Int) : Int :=
  if 0 ≤ u ∧ 0 ≤ v then (if u = 0 ∧ v = 0 then 0 else 1)
  else if u ≤ 0 ∧ v ≤ 0 then -1
  else if 0 < u then (if u * u > 2 * v * v then 1 else -1)
  else (if 2 * v * v > u * u then 1 else -1)

/-- (U, V, T) for α = a + b√2 at index k. -/
def theta (a b k : Int) : Int × Int × Int :=
  let (p, q) := epsPow k
  (2 * (a * q + b * p), a * p + 2 * b * q, a * q + b * p)

def floorOK (a b k n : Int) : Bool :=
  let (U, V, _) := theta a b k
  decide (0 ≤ sgn2 (U + 2 - 4 * n) V) && decide (sgn2 (U + 2 - 4 * n - 4) V < 0)

def allFloors (a b k : Int) : List Int → Bool
  | [] => true
  | n :: ns => floorOK a b k n && allFloors a b (k + 1) ns

def digits : List Int → List Int
  | n2 :: n1 :: n0 :: rest => (n0 - 2 * n1 - n2) :: digits (n1 :: n0 :: rest)
  | _ => []

def trimL : List Int → List Int
  | 0 :: xs => trimL xs
  | xs => xs
def trim (xs : List Int) : List Int := (trimL (trimL xs).reverse).reverse

/-- the i-th entry from the end of ns equals T at its index. -/
def lastT (a b k : Int) (ns : List Int) (i : Nat) : Bool :=

  | some n => decide (n = (theta a b (k + ((ns.length - 1 - i : Nat) : Int))).2.2)

def cert (a b k0 : Int) (ns w : List Int) : Bool :=

  lastT a b k0 ns 0 && lastT a b k0 ns 1 && lastT a b k0 ns 2 && decide (trim (digits ns) = w)
